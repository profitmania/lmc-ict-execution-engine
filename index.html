// Strict Engulfing Calculations

bool isBearEngulf = close < open and close[1] > open[1] and (engulfMode == "Body Engulf" ? close < open[1] : close < low[1])

bool passDisplacement = not useDisplacement or isDisplacementBar or isDisplacementBar[1]

bool buyTrigger  = exEnabled and exState == 0 and sslArmed and isBullEngulf and passDisplacement
bool sellTrigger = exEnabled and exState == 0 and bslArmed and isBearEngulf and passDisplacement

if buyTrigger
    float candidateEntry = math.round(close / syminfo.mintick) * syminfo.mintick
    float refLow         = math.min(low, low[1])
    float candidateStop  = math.round((refLow - exStopTicks * syminfo.mintick) / syminfo.mintick) * syminfo.mintick
    float riskPts        = candidateEntry - candidateStop
    
    if riskPts > 0 and riskPts <= atrValue * maxStopATRMult
        exState      := 4
        exDir        := 1
        exEntryBar   := bar_index
        exEntryPrice := candidateEntry
        exStopLoss   := candidateStop
        exTakeProfit := math.round((exEntryPrice + riskPts * exTargetR) / syminfo.mintick) * syminfo.mintick
        float riskPerContract = riskPts * exPointValue
        exContracts  := riskPerContract > 0 ? int(math.floor(exMaxRisk / riskPerContract)) : 0
        sslArmed     := false
        
        label.delete(histSweepLabel)
        histSweepLabel   := activeSweepLabel
        activeSweepLabel := na
    else if riskPts > atrValue * maxStopATRMult
        sslArmed     := false

if sellTrigger
    float candidateEntry = math.round(close / syminfo.mintick) * syminfo.mintick
    float refHigh        = math.max(high, high[1])
    float candidateStop  = math.round((refHigh + exStopTicks * syminfo.mintick) / syminfo.mintick) * syminfo.mintick
    float riskPts        = candidateStop - candidateEntry
    
    if riskPts > 0 and riskPts <= atrValue * maxStopATRMult
        exState      := 4
        exDir        := -1
        exEntryBar   := bar_index
        exEntryPrice := candidateEntry
        exStopLoss   := candidateStop
        exTakeProfit := math.round((exEntryPrice - riskPts * exTargetR) / syminfo.mintick) * syminfo.mintick
        float riskPerContract = riskPts * exPointValue
        exContracts  := riskPerContract > 0 ? int(math.floor(exMaxRisk / riskPerContract)) : 0
        bslArmed     := false
        
        label.delete(histSweepLabel)
        histSweepLabel   := activeSweepLabel
        activeSweepLabel := na
    else if riskPts > atrValue * maxStopATRMult
        bslArmed     := false

    bool hitTP = exDir == 1 ? high >= exTakeProfit : low <= exTakeProfit
    
    if hitSL or hitTP
